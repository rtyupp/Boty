import os
import logging
import yfinance as yf
import mplfinance as mpf
from aiohttp import web
from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup
from telegram.ext import Application, CommandHandler, ContextTypes, CallbackQueryHandler

# جلب البيانات الحساسة من متغيرات الخادم (آمن 100%)
BOT_TOKEN = os.environ.get("BOT_TOKEN")
CHAT_ID = os.environ.get("CHAT_ID")
SYMBOL = "GC=F"

logging.basicConfig(format='%(asctime)s - %(name)s - %(levelname)s - %(message)s', level=logging.INFO)

# ================= دوال التحليل =================
def get_gold_data():
    try:
        data = yf.download(SYMBOL, period='5d', interval='15m', progress=False)
        if data.empty: return None
        data['EMA_20'] = data['Close'].ewm(span=20, adjust=False).mean()
        data['EMA_50'] = data['Close'].ewm(span=50, adjust=False).mean()
        return data
    except Exception as e:
        logging.error(f"Data Error: {e}")
        return None

def generate_chart():
    data = get_gold_data()
    if data is None: return None
    add_plots = [
        mpf.make_addplot(data['EMA_20'], color='blue', width=1),
        mpf.make_addplot(data['EMA_50'], color='orange', width=1)
    ]
    mpf.plot(data, type='candle', addplot=add_plots, title='Gold (XAU/USD) 15m',
             style='charles', savefig='gold_chart.png', figsize=(10, 6))
    return 'gold_chart.png'

# ================= دوال تيليجرام =================
async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    keyboard = [[InlineKeyboardButton("📊 عرض شارت الذهب", callback_data='show_chart')]]
    await update.message.reply_text("🤖 بوت الذهب يعمل 24/7!\nاضغط على الزر أدناه.", reply_markup=InlineKeyboardMarkup(keyboard))

async def button_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    
    if query.data == 'show_chart':
        await query.edit_message_text("⏳ جاري رسم الشارت...")
        chart_path = generate_chart()
        
        if chart_path:
            keyboard = [[InlineKeyboardButton("🔄 تحديث", callback_data='show_chart')]]
            with open(chart_path, 'rb') as photo:
                await context.bot.send_photo(
                    chat_id=query.message.chat_id,
                    photo=photo,
                    caption="📈 شارت الذهب المباشر",
                    reply_markup=InlineKeyboardMarkup(keyboard)
                )
        else:
            await query.edit_message_text("❌ فشل في جلب البيانات.")

# ================= إعداد الخادم والبوت =================
async def health_check(request):
    """هذه النقطة هي ما سيقوم UptimeRobot بزيارتها لإبقاء البوت مستيقظاً"""
    return web.Response(text="Bot is Alive and Running 24/7 ✅")

async def on_startup(app):
    """تشغيل بوت تيليجرام عند بدء الخادم"""
    app['bot_app'] = Application.builder().token(BOT_TOKEN).build()
    app['bot_app'].add_handler(CommandHandler("start", start))
    app['bot_app'].add_handler(CallbackQueryHandler(button_handler))
    
    await app['bot_app'].initialize()
    await app['bot_app'].start()
    
    # تشغيل البوت في الخلفية
    import asyncio
    asyncio.create_task(app['bot_app'].updater.start_polling(allowed_updates=Update.ALL_TYPES))

async def on_cleanup(app):
    """إيقاف نظيف عند إغلاق الخادم"""
    await app['bot_app'].stop()
    await app['bot_app'].shutdown()

# تشغيل التطبيق
if __name__ == '__main__':
    web_app = web.Application()
    web_app.on_startup.append(on_startup)
    web_app.on_cleanup.append(on_cleanup)
    web_app.router.add_get('/', health_check) # رابط الفحص
    
    port = int(os.environ.get("PORT", 8080))
    print(f"🚀 Starting server on port {port}...")
    web.run_app(web_app, host='0.0.0.0', port=port)
